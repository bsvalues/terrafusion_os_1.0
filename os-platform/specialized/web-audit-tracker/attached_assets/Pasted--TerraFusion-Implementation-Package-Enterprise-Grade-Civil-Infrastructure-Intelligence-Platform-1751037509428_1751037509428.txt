### Python API Usage
```python
from terrafusion import TerraFusionClient

# Initialize client
client = TerraFusionClient(api_key="your-api-key")

# Single property assessment
result = await client.assess_property({
    "address": "123 Main St",
    "city": "Anytown",
    "state": "CA"
})
print(f"Valuation: ${result.valuation:,.2f}")

# Batch processing
properties = [
    {"address": "123 Main St", "city": "Anytown", "state": "CA"},
    {"address": "456 Oak Ave", "city": "Anytown", "state": "CA"}
]
results = await client.batch_assess(properties)
for result in results:
    print(f"{result.address}: ${result.valuation:,.2f}")

# Real-time monitoring
async for event in client.monitor_infrastructure():
    if event.type == "sensor_alert":
        print(f"Alert: {event.message}")
```
