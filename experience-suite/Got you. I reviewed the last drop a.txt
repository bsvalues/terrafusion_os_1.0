Got you. I reviewed the last drop and tightened everything. A few things needed fixing to be “can’t-break-it” solid:

**What I fixed vs. v6**

* **Path aliases** → removed `@/…` usage so the UI compiles without tsconfig alias tweaks.
* **MSAL location** → AuthProvider is now placed under `ui/src/auth/msal/…` (where the admin shell expects it).
* **Style Dictionary filter** → removed the invalid `isBase` filter; now uses attribute filters (works out-of-the-box).
* **Admin policy wiring** → added the `AdminOnly` policy and DbContext registration to the Program.cs patch.
* **CSP duplication** → rely on CloudFront Response Headers Policy (removed the redundant Kong CSP header).
* **Kong tenant header** → still included (optional) and guarded behind a script.
* **Idempotent bootstrap** → preflight checks, dry-run, and clear outputs.

Below is a **single self-extracting, idempotent bootstrap**. It recreates the entire **Suite v6.1** layout locally (no zips), with all fixes baked in.

# One-shot installer (v6.1)

Save as `bootstrap-terrafusion-suite-v6.1.sh`, then:

```bash
bash bootstrap-terrafusion-suite-v6.1.sh
cd terrafusion-suite-v6.1
./verify-suite.sh
# When kube context is ready:
./enable-v6.sh                # installs cert-manager, issuers, Helmfile stack, builds tokens
./helmfile-apply-env.sh staging   # or production
```

> If you want a different base directory, run with `DEST=/your/path bash bootstrap-terrafusion-suite-v6.1.sh`

````bash
#!/usr/bin/env bash
set -euo pipefail

DEST="${DEST:-${PWD}}"
ROOT="${DEST%/}/terrafusion-suite-v6.1"

green(){ printf "\033[32m%s\033[0m\n" "$*"; }
yellow(){ printf "\033[33m%s\033[0m\n" "$*"; }
red(){ printf "\033[31m%s\033[0m\n" "$*"; }
w(){ # write file with heredoc
  local path="$1"; shift
  mkdir -p "$(dirname "${ROOT}/${path}")"
  cat > "${ROOT}/${path}" <<'EOF'
'"$*"'EOF'
  echo "✓ ${path}"
}

echo
green ">>> TerraFusion Suite v6.1 — Self-Extracting Bootstrap"
echo "Target: ${ROOT}"
mkdir -p "${ROOT}"

import React, { createContext, useContext, useMemo } from "react";
import { PublicClientApplication, AccountInfo, InteractionRequiredAuthError } from "@azure/msal-browser";
import { MsalProvider } from "@azure/msal-react";

const msalConfig = {
  auth: {
    clientId: import.meta.env.VITE_AZURE_AD_CLIENT_ID!,
    authority: `https://login.microsoftonline.com/${import.meta.env.VITE_AZURE_AD_TENANT_ID}`,
    redirectUri: import.meta.env.VITE_AUTH_REDIRECT_URI || window.location.origin + "/admin/callback"
  },
  cache: { cacheLocation: "localStorage" }
};
const pca = new PublicClientApplication(msalConfig);
export function AzureAuthRoot({ children }: { children: React.ReactNode }) { return <MsalProvider instance={pca}>{children}</MsalProvider>; }

type AuthCtx = { account?: AccountInfo | null; hasGroup: (gid:string)=>boolean; acquireToken:(scopes?:string[])=>Promise<string>; };
const Ctx = createContext<AuthCtx>({ hasGroup: ()=>false, acquireToken: async()=>"" });
export const useAuth = () => useContext(Ctx);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const accounts = pca.getAllAccounts(); const account = accounts[0];
  const value = useMemo<AuthCtx>(() => ({
    account,
    hasGroup: (gid:string) => { const raw = (account as any)?.idTokenClaims?.groups as string[] | undefined; return !!raw?.includes(gid); },
    acquireToken: async (scopes: string[] = ["openid","profile","email"]) => {
      try { const res = await pca.acquireTokenSilent({ account: account!, scopes }); return res.accessToken || res.idToken; }
      catch(e){ if (e instanceof InteractionRequiredAuthError) { const res = await pca.acquireTokenPopup({ scopes }); return res.accessToken || res.idToken; } throw e; }
    }
  }), [account]);
  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}
export function RequireAdmin({ children, adminGroupId }:{children:React.ReactNode; adminGroupId:string}){
  const { hasGroup } = useAuth();
  if (!hasGroup(adminGroupId)) return <div className="p-6">Admin access required.</div>;
  return <>{children}</>;
}

w ui/src/brand/countyTheme.ts '
export type County = "Benton" | "Yakima" | string;
export function applyCountyTheme(county: County) {
  ensure("tf-base","/brand/tokens-base.css");
  const map: Record<string,string> = { Benton:"/brand/tokens-benton.css", Yakima:"/brand/tokens-yakima.css" };
  if (map[county]) ensure("tf-county", map[county]); else remove("tf-county");
}
function ensure(id:string, href:string){ let l=document.getElementById(id) as HTMLLinkElement|null; if(!l){l=document.createElement("link"); l.id=id; l.rel="stylesheet"; l.href=href; document.head.appendChild(l);} else if(l.href!==href){l.href=href;} }
function remove(id:string){ const l=document.getElementById(id); if(l) l.remove(); }
'

  const [county, setCounty] = React.useState<string>(() => localStorage.getItem("tf-county") || "Benton");

  return <CountyContext.Provider value={{ county, setCounty }}>{children}</CountyContext.Provider>;
}

using System.Text.RegularExpressions;
using Microsoft.AspNetCore.Http;
using Microsoft.Extensions.Primitives;
using TerraFusion.Core.Multitenancy;
namespace TerraFusion.API.Middleware {
  public class TenantResolutionMiddleware: IMiddleware {
    private readonly TenantContext _ctx;
    private static readonly Regex Subdomain=new(@"^([a-z0-9-]+)\.", System.Text.RegularExpressions.RegexOptions.IgnoreCase|System.Text.RegularExpressions.RegexOptions.Compiled);
    public TenantResolutionMiddleware(TenantContext ctx){ _ctx=ctx; }
    public async System.Threading.Tasks.Task InvokeAsync(HttpContext http, RequestDelegate next){
      if (http.Request.Headers.TryGetValue("X-TF-Tenant", out StringValues h) && !StringValues.IsNullOrEmpty(h)) _ctx.Set(h!);
      else if (http.Request.Cookies.TryGetValue("tf_tenant", out var c) && !string.IsNullOrWhiteSpace(c)) _ctx.Set(c);
      else { var host=http.Request.Host.Host; var m=Subdomain.Match(host); if(m.Success) _ctx.Set(m.Groups[1].Value); }
      if (!http.Request.Cookies.ContainsKey("tf_tenant"))
        http.Response.Cookies.Append("tf_tenant", _ctx.GetTenantId(), new CookieOptions{ SameSite=SameSiteMode.Lax, HttpOnly=false, Secure=true, IsEssential=true, MaxAge=System.TimeSpan.FromDays(365)});
      await next(http);
    }
  }
}
````
