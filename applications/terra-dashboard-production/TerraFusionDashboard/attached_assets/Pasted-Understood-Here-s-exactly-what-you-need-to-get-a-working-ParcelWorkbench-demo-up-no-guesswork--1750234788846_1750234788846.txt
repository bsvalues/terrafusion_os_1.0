## **src/App.jsx**

```jsx
import React, { useState } from "react";
import axios from "axios";

function App() {
  const [query, setQuery] = useState("");
  const [parcels, setParcels] = useState([]);
  const [loading, setLoading] = useState(false);

  const searchParcels = async () => {
    setLoading(true);
    const res = await axios.post("http://localhost:8000/api/parcels/search", { query });
    setParcels(res.data.data);
    setLoading(false);
  };

  return (
    <div className="max-w-2xl mx-auto p-4">
      <h1 className="text-2xl mb-4 font-bold">ParcelWorkbench Demo</h1>
      <input
        className="border rounded p-2 w-full mb-2"
        placeholder="Search parcels"
        value={query}
        onChange={e => setQuery(e.target.value)}
      />
      <button
        className="bg-blue-500 text-white px-4 py-2 rounded"
        onClick={searchParcels}
        disabled={loading}
      >
        {loading ? "Searching..." : "Search"}
      </button>
      <ul className="mt-4">
        {parcels.map(p => (
          <li key={p.id} className="border-b py-2">
            <b>{p.parcelNumber}</b> – {p.owner} (${p.assessedValue})
          </li>
        ))}
      </ul>
    </div>
  );
}
export default App;
```

## **main.py**

```python
from fastapi import FastAPI, Request
from pydantic import BaseModel
from typing import List, Optional
from uuid import uuid4
from fastapi.middleware.cors import CORSMiddleware

app = FastAPI()
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],  # In prod, restrict this!
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

class Parcel(BaseModel):
    id: str
    parcelNumber: str
    owner: str
    assessedValue: float

# Fake in-memory DB
db = [
    Parcel(id=str(uuid4()), parcelNumber="1001", owner="John Doe", assessedValue=215000),
    Parcel(id=str(uuid4()), parcelNumber="1002", owner="Jane Smith", assessedValue=322000)
]

class SearchRequest(BaseModel):
    query: Optional[str] = ""

@app.post("/api/parcels/search")
def search_parcels(req: SearchRequest):
    q = req.query.lower()
    return {
        "success": True,
        "data": [
            p for p in db
            if q in p.parcelNumber.lower() or q in p.owner.lower()
        ]
    }

@app.get("/health")
def health():
    return {"status": "ok"}
```

# 4. **Start Everything (Dev Mode)**

```bash
# 1. Start the API
cd apps/api-server
pip install -r requirements.txt
uvicorn main:app --reload

# 2. Start the Dashboard
cd ../dashboard-frontend
npm install
npm start
```

* Open [http://localhost:3000](http://localhost:3000)
* Type a name (“John”) or parcel (“1001”) and search.
