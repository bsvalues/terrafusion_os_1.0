# Terrafusion Platform - Divine Container Configuration
# Multi-stage build for Tesla-grade optimization
# ================================================================

# Stage 1: Build Environment
FROM python:3.11-slim as builder

# Set divine environment variables
ENV PYTHONUNBUFFERED=1 \
    PYTHONDONTWRITEBYTECODE=1 \
    PIP_NO_CACHE_DIR=1 \
    PIP_DISABLE_PIP_VERSION_CHECK=1

# Install system dependencies
RUN apt-get update && apt-get install -y \
    build-essential \
    libpq-dev \
    libssl-dev \
    libffi-dev \
    cargo \
    && rm -rf /var/lib/apt/lists/*

# Create application user (security best practice)
RUN groupadd --gid 1000 terrafusion \
    && useradd --uid 1000 --gid terrafusion --shell /bin/bash --create-home terrafusion

# Set work directory
WORKDIR /app

# Copy and install Python dependencies
COPY requirements.txt .
RUN pip install --user --no-cache-dir -r requirements.txt

# Stage 2: Production Environment
FROM python:3.11-slim as production

# Install runtime dependencies only
RUN apt-get update && apt-get install -y \
    libpq5 \
    curl \
    && rm -rf /var/lib/apt/lists/* \
    && apt-get clean

# Create application user
RUN groupadd --gid 1000 terrafusion \
    && useradd --uid 1000 --gid terrafusion --shell /bin/bash --create-home terrafusion

# Copy Python packages from builder
COPY --from=builder /home/terrafusion/.local /home/terrafusion/.local

# Set work directory
WORKDIR /app

# Copy application code
COPY --chown=terrafusion:terrafusion . .

# Set Python path
ENV PATH=/home/terrafusion/.local/bin:$PATH

# Create necessary directories
RUN mkdir -p /app/logs /app/data /app/ssl \
    && chown -R terrafusion:terrafusion /app

# Switch to non-root user
USER terrafusion

# Health check with divine precision
HEALTHCHECK --interval=30s --timeout=10s --start-period=60s --retries=3 \
    CMD curl -f http://localhost:8000/api/health || exit 1

# Expose divine port
EXPOSE 8000

# Command with proper signal handling
CMD ["python", "-m", "uvicorn", "terrafusion_platform:app", "--host", "0.0.0.0", "--port", "8000", "--workers", "4"]
