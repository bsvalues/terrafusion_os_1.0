# Logging functions
log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {

}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

log_step() {
    echo -e "${BLUE}[STEP]${NC} $1"

log_success() {
    echo -e "${GREEN}[SUCCESS]${NC} $1"

# Configuration validation
check_prerequisites() {
    log_step "Checking prerequisites..."
    
    # Check required tools
    local tools=("docker" "kubectl" "helm" "terraform" "git")
    for tool in "${tools[@]}"; do
        if ! command -v "$tool" &> /dev/null; then
            log_error "$tool is not installed. Please install it first."
            exit 1
        fi
        log_info "✓ $tool found"
    done
    
    # Check environment variables
    local env_vars=("DB_PASSWORD" "JWT_SECRET" "REDIS_PASSWORD" "SSL_CERT_PATH")
    for var in "${env_vars[@]}"; do
        if [[ -z "${!var:-}" ]]; then
            log_error "Environment variable $var is not set"
            exit 1
        fi
        log_info "✓ $var configured"
    done
    
    # Check SSL certificates
    if [[ ! -f "$SSL_CERT_PATH/cert.pem" ]] || [[ ! -f "$SSL_CERT_PATH/key.pem" ]]; then
        log_error "SSL certificates not found in $SSL_CERT_PATH"
        exit 1
    fi
    log_info "✓ SSL certificates found"
    
    log_success "Prerequisites check completed"
}

# Database initialization
setup_database() {

    kubectl wait --for=condition=ready pod -l app=postgresql -n terrafusion-production --timeout=300s
    
    # Database schema creation
    cat > schema.sql << 'EOF'
-- Terrafusion Production Database Schema
-- Optimized for 94,149 Benton County Washington parcels

-- Enable required extensions
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "postgis";
CREATE EXTENSION IF NOT EXISTS "pg_stat_statements";
CREATE EXTENSION IF NOT EXISTS "pg_trgm";

-- Create schemas
CREATE SCHEMA IF NOT EXISTS terra_core;
CREATE SCHEMA IF NOT EXISTS terra_agent;
CREATE SCHEMA IF NOT EXISTS terra_flow;
CREATE SCHEMA IF NOT EXISTS terra_sync;
CREATE SCHEMA IF NOT EXISTS county_plugins;

-- Set search path
SET search_path TO terra_core, public;

-- Core properties table (optimized for 94,149+ parcels)
CREATE TABLE properties (
    parcel_id VARCHAR(50) PRIMARY KEY,
    parcel_number VARCHAR(30) NOT NULL,
    account_number VARCHAR(30),
    owner_name VARCHAR(200) NOT NULL,
    situs_address VARCHAR(300) NOT NULL,
    legal_description TEXT,
    property_class VARCHAR(10) NOT NULL,
    land_use_code VARCHAR(10),
    land_value DECIMAL(12,2) NOT NULL DEFAULT 0,
    improvement_value DECIMAL(12,2) NOT NULL DEFAULT 0,
    total_value DECIMAL(12,2) NOT NULL DEFAULT 0,
    assessed_value DECIMAL(12,2) NOT NULL DEFAULT 0,
    tax_year INTEGER NOT NULL,
    square_footage INTEGER,
    lot_size DECIMAL(10,2),
    year_built INTEGER,
    bedrooms INTEGER,
    bathrooms DECIMAL(3,1),
    geometry GEOMETRY(POLYGON, 4326),
    last_sale_date DATE,
    last_sale_price DECIMAL(12,2),
    exemptions JSONB DEFAULT '[]'::jsonb,
    metadata JSONB DEFAULT '{}'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    deleted_at TIMESTAMP WITH TIME ZONE
);

-- Performance indexes for 94,149 parcels
CREATE INDEX idx_properties_parcel_number ON properties(parcel_number);
CREATE INDEX idx_properties_owner_name_gin ON properties USING gin(owner_name gin_trgm_ops);
CREATE INDEX idx_properties_situs_address_gin ON properties USING gin(situs_address gin_trgm_ops);
CREATE INDEX idx_properties_property_class ON properties(property_class);
CREATE INDEX idx_properties_total_value ON properties(total_value);
CREATE INDEX idx_properties_tax_year ON properties(tax_year);
CREATE INDEX idx_properties_geometry ON properties USING gist(geometry);
CREATE INDEX idx_properties_last_sale_date ON properties(last_sale_date);
CREATE INDEX idx_properties_deleted_at ON properties(deleted_at) WHERE deleted_at IS NULL;

-- Assessments table
CREATE TABLE assessments (
    assessment_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    parcel_id VARCHAR(50) NOT NULL REFERENCES properties(parcel_id),
    assessor_id UUID NOT NULL,
    assessment_date TIMESTAMP WITH TIME ZONE NOT NULL,
    assessment_type VARCHAR(30) NOT NULL CHECK (assessment_type IN ('annual', 'revaluation', 'appeal', 'new_construction')),
    land_value DECIMAL(12,2) NOT NULL,
    improvement_value DECIMAL(12,2) NOT NULL,
    total_value DECIMAL(12,2) NOT NULL,
    methodology TEXT NOT NULL,
    confidence_score DECIMAL(3,2) CHECK (confidence_score >= 0 AND confidence_score <= 1),
    ai_analysis JSONB,
    status VARCHAR(20) NOT NULL DEFAULT 'draft' CHECK (status IN ('draft', 'pending_review', 'approved', 'rejected')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_assessments_parcel_id ON assessments(parcel_id);
CREATE INDEX idx_assessments_assessor_id ON assessments(assessor_id);
CREATE INDEX idx_assessments_date ON assessments(assessment_date);
CREATE INDEX idx_assessments_status ON assessments(status);

-- AI agent results table
CREATE TABLE terra_agent.agent_results (
    result_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    agent_type VARCHAR(50) NOT NULL,
    operation VARCHAR(50) NOT NULL,
    parcel_id VARCHAR(50) REFERENCES terra_core.properties(parcel_id),
    input_data JSONB NOT NULL,
    output_data JSONB NOT NULL,
    confidence DECIMAL(3,2),
    processing_time INTEGER NOT NULL, -- milliseconds
    success BOOLEAN NOT NULL DEFAULT true,
    error_message TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_agent_results_agent_type ON terra_agent.agent_results(agent_type);
CREATE INDEX idx_agent_results_parcel_id ON terra_agent.agent_results(parcel_id);
CREATE INDEX idx_agent_results_created_at ON terra_agent.agent_results(created_at);

-- Audit log table for compliance
CREATE TABLE audit_log (
    log_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    table_name VARCHAR(100) NOT NULL,
    record_id VARCHAR(100) NOT NULL,
    operation VARCHAR(10) NOT NULL CHECK (operation IN ('INSERT', 'UPDATE', 'DELETE')),
    old_values JSONB,
    new_values JSONB,
    user_id UUID,
    timestamp TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    ip_address INET,
    user_agent TEXT
);

CREATE INDEX idx_audit_log_table_record ON audit_log(table_name, record_id);
CREATE INDEX idx_audit_log_timestamp ON audit_log(timestamp);
CREATE INDEX idx_audit_log_user_id ON audit_log(user_id);

-- Performance statistics view
CREATE VIEW performance_stats AS
SELECT 
    COUNT(*) as total_properties,
    AVG(total_value) as avg_property_value,
    SUM(total_value) as total_assessed_value,
    COUNT(CASE WHEN last_sale_date >= CURRENT_DATE - INTERVAL '1 year' THEN 1 END) as recent_sales,
    AVG(CASE WHEN last_sale_date >= CURRENT_DATE - INTERVAL '1 year' THEN last_sale_price END) as avg_sale_price
FROM properties 
WHERE deleted_at IS NULL;

-- IAAO compliance monitoring view
CREATE VIEW iaao_compliance AS
WITH ratio_study AS (
    SELECT 
        p.parcel_id,
        p.assessed_value::decimal / NULLIF(p.last_sale_price, 0) as ratio
    FROM properties p
    WHERE p.last_sale_date >= CURRENT_DATE - INTERVAL '2 years'
      AND p.last_sale_price > 0
      AND p.deleted_at IS NULL
),
stats AS (
    SELECT 
        AVG(ratio) as mean_ratio,
        PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY ratio) as median_ratio,
        STDDEV(ratio) as std_dev,
        COUNT(*) as sample_size
    FROM ratio_study
    WHERE ratio BETWEEN 0.5 AND 2.0
)
SELECT 
    mean_ratio,
    median_ratio,
    std_dev,
    sample_size,
    (ABS(mean_ratio - median_ratio) / median_ratio * 100) as coefficient_of_dispersion,
    CASE 
        WHEN (ABS(mean_ratio - median_ratio) / median_ratio * 100) <= 15 THEN 'Compliant'
        ELSE 'Non-Compliant'
    END as iaao_status
FROM stats;

-- Trigger for audit logging
CREATE OR REPLACE FUNCTION audit_trigger() RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'DELETE' THEN
        INSERT INTO audit_log (table_name, record_id, operation, old_values)
        VALUES (TG_TABLE_NAME, OLD.parcel_id, TG_OP, row_to_json(OLD));
        RETURN OLD;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO audit_log (table_name, record_id, operation, old_values, new_values)
        VALUES (TG_TABLE_NAME, NEW.parcel_id, TG_OP, row_to_json(OLD), row_to_json(NEW));
        RETURN NEW;
    ELSIF TG_OP = 'INSERT' THEN
        INSERT INTO audit_log (table_name, record_id, operation, new_values)
        VALUES (TG_TABLE_NAME, NEW.parcel_id, TG_OP, row_to_json(NEW));
        RETURN NEW;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- Apply audit triggers
CREATE TRIGGER properties_audit_trigger
    AFTER INSERT OR UPDATE OR DELETE ON properties
    FOR EACH ROW EXECUTE FUNCTION audit_trigger();

CREATE TRIGGER assessments_audit_trigger
    AFTER INSERT OR UPDATE OR DELETE ON assessments
    FOR EACH ROW EXECUTE FUNCTION audit_trigger();

-- Update timestamp trigger
CREATE OR REPLACE FUNCTION update_timestamp() RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER properties_update_timestamp
    BEFORE UPDATE ON properties
    FOR EACH ROW EXECUTE FUNCTION update_timestamp();

CREATE TRIGGER assessments_update_timestamp
    BEFORE UPDATE ON assessments
    FOR EACH ROW EXECUTE FUNCTION update_timestamp();
EOF

    # Execute schema
    kubectl exec -i $(kubectl get pods -l app=postgresql -n terrafusion-production -o jsonpath="{.items[0].metadata.name}") -n terrafusion-production -- psql -U terrafusion -d terrafusion_prod < schema.sql
    
    log_success "Database schema created successfully"
}

    cat > monitoring/prometheus.yml << 'EOF'

  evaluation_interval: 15s

  - "terrafusion_rules.yml"

    - static_configs:

    static_configs:
      - targets: ['terrafusion-api-service:3000']

  - job_name: 'terrafusion-agents'
    static_configs:
      - targets: ['terrafusion-agents-service:8000']
    metrics_path: '/metrics'
    scrape_interval: 15s

  - job_name: 'postgresql'
    static_configs:
      - targets: ['postgresql-service:9187']
    scrape_interval: 30s

  - job_name: 'redis'
    static_configs:
      - targets: ['redis-service:9121']
    scrape_interval: 30s

  - job_name: 'kubernetes-nodes'
    kubernetes_sd_configs:
      - role: node
    relabel_configs:
      - source_labels: [__address__]
        regex: '(.*):10250'
        target_label: __address__
        replacement: '${1}:9100'
EOF

    # Alerting rules
    cat > monitoring/terrafusion_rules.yml << 'EOF'
groups:
  - name: terrafusion.rules
    rules:
      # API Health
      - alert: TerraFusionAPIDown
        expr: up{job="terrafusion-api"} == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "Terrafusion API is down"
          description: "Terrafusion API has been down for more than 1 minute."

      # High Response Time
      - alert: TerraFusionHighResponseTime
        expr: histogram_quantile(0.95, rate(http_request_duration_seconds_bucket{job="terrafusion-api"}[5m])) > 0.5
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Terrafusion API high response time"
          description: "95th percentile response time is {{ $value }}s"

      # High Error Rate
      - alert: TerraFusionHighErrorRate
        expr: rate(http_requests_total{job="terrafusion-api",status=~"5.."}[5m]) / rate(http_requests_total{job="terrafusion-api"}[5m]) > 0.05
        for: 2m
        labels:
          severity: critical
        annotations:
          summary: "Terrafusion API high error rate"
          description: "Error rate is {{ $value | humanizePercentage }}"

      # AI Agents Down
      - alert: TerraFusionAgentsDown
        expr: up{job="terrafusion-agents"} == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "Terrafusion AI Agents are down"
          description: "AI Agents service has been down for more than 1 minute."

      # Database Connection Issues
      - alert: PostgreSQLDown
        expr: up{job="postgresql"} == 0
        for: 30s
        labels:
          severity: critical
        annotations:
          summary: "PostgreSQL is down"
          description: "PostgreSQL database has been down for more than 30 seconds."

      # High Database Connections
      - alert: PostgreSQLHighConnections
        expr: pg_stat_database_numbackends / pg_settings_max_connections > 0.8
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "PostgreSQL high connection usage"
          description: "Database connections are at {{ $value | humanizePercentage }} of maximum."

      # Disk Space
      - alert: DiskSpaceHigh
        expr: (node_filesystem_size_bytes - node_filesystem_avail_bytes) / node_filesystem_size_bytes > 0.85
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Disk space usage high"
          description: "Disk usage is above 85% on {{ $labels.instance }}"

      # Memory Usage
      - alert: MemoryUsageHigh
        expr: (node_memory_MemTotal_bytes - node_memory_MemAvailable_bytes) / node_memory_MemTotal_bytes > 0.9
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Memory usage high"
          description: "Memory usage is above 90% on {{ $labels.instance }}"
EOF

    # Deploy monitoring stack with Helm
    helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
    helm repo add grafana https://grafana.github.io/helm-charts
    helm repo update
    
    # Install Prometheus
    helm install prometheus prometheus-community/kube-prometheus-stack \
        --namespace terrafusion-production \
        --values monitoring/prometheus-values.yml \
        --wait
    
    # Install Grafana dashboards
    kubectl apply -f monitoring/grafana-dashboards.yml -n terrafusion-production
    
    log_success "Monitoring and alerting setup completed"
}

# Data migration from existing systems
migrate_data() {
    log_step "Migrating data from existing systems..."
    
    # Create migration script for Benton County data
    cat > migration/migrate_benton_county.py << 'EOF'
#!/usr/bin/env python3
"""
Terrafusion Data Migration Script
Migrates Benton County Washington data (94,149 parcels) to Terrafusion platform
"""

import csv
import psycopg2
import logging
from datetime import datetime
from typing import Dict, List, Optional
import json

# Configure logging
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)

class BentonCountyMigrator:
    def __init__(self, db_config: Dict[str, str]):
        self.db_config = db_config
        self.conn = None
        self.cursor = None
        
    def connect(self):
        """Establish database connection"""
        try:
            self.conn = psycopg2.connect(**self.db_config)
            self.cursor = self.conn.cursor()
            logger.info("Database connection established")
        except Exception as e:
            logger.error(f"Database connection failed: {e}")
            raise
    
    def migrate_properties(self, csv_files: Dict[str, str]) -> int:
        """Migrate property data from CSV files"""
        logger.info("Starting property data migration...")
        
        # Read property values
        property_values = {}
        with open(csv_files['property_val'], 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            for row in reader:
                property_values[row['parcel_id']] = row
        
        # Read addresses
        addresses = {}
        with open(csv_files['situs'], 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            for row in reader:
                addresses[row['parcel_id']] = row
        
        # Read accounts
        accounts = {}
        with open(csv_files['account'], 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            for row in reader:
                accounts[row['parcel_id']] = row
        
        # Read improvements
        improvements = {}
        with open(csv_files['imprv'], 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            for row in reader:
                if row['parcel_id'] not in improvements:
                    improvements[row['parcel_id']] = []
                improvements[row['parcel_id']].append(row)
        
        # Process and insert data
        insert_count = 0
        batch_size = 1000
        batch_data = []
        
        for parcel_id, prop_val in property_values.items():
            try:
                # Combine data from different sources
                address_data = addresses.get(parcel_id, {})
                account_data = accounts.get(parcel_id, {})
                improvement_data = improvements.get(parcel_id, [])
                
                # Prepare record
                record = {
                    'parcel_id': parcel_id,
                    'parcel_number': prop_val.get('parcel_number', ''),
                    'account_number': account_data.get('account_number', ''),
                    'owner_name': account_data.get('owner_name', ''),
                    'situs_address': address_data.get('situs_address', ''),
                    'legal_description': prop_val.get('legal_description', ''),
                    'property_class': prop_val.get('property_class', ''),
                    'land_use_code': prop_val.get('land_use_code', ''),
                    'land_value': float(prop_val.get('land_value', 0) or 0),
                    'improvement_value': float(prop_val.get('improvement_value', 0) or 0),
                    'total_value': float(prop_val.get('total_value', 0) or 0),
                    'assessed_value': float(prop_val.get('assessed_value', 0) or 0),
                    'tax_year': int(prop_val.get('tax_year', 2025)),
                    'square_footage': int(prop_val.get('square_footage', 0) or 0) if prop_val.get('square_footage') else None,
                    'lot_size': float(prop_val.get('lot_size', 0) or 0) if prop_val.get('lot_size') else None,
                    'year_built': int(prop_val.get('year_built', 0) or 0) if prop_val.get('year_built') else None,
                    'bedrooms': int(prop_val.get('bedrooms', 0) or 0) if prop_val.get('bedrooms') else None,
                    'bathrooms': float(prop_val.get('bathrooms', 0) or 0) if prop_val.get('bathrooms') else None,
                    'last_sale_date': prop_val.get('last_sale_date') if prop_val.get('last_sale_date') else None,
                    'last_sale_price': float(prop_val.get('last_sale_price', 0) or 0) if prop_val.get('last_sale_price') else None,
                    'exemptions': json.dumps([]),
                    'metadata': json.dumps({
                        'source': 'benton_county_migration',
                        'migration_date': datetime.now().isoformat(),
                        'improvements': improvement_data
                    })
                }
                
                batch_data.append(record)
                
                # Insert batch when full
                if len(batch_data) >= batch_size:
                    self._insert_batch(batch_data)
                    insert_count += len(batch_data)
                    logger.info(f"Migrated {insert_count} properties...")
                    batch_data = []
                    
            except Exception as e:
                logger.error(f"Error processing parcel {parcel_id}: {e}")
                continue
        
        # Insert remaining batch
        if batch_data:
            self._insert_batch(batch_data)
            insert_count += len(batch_data)
        
        self.conn.commit()
        logger.info(f"Migration completed. Total properties migrated: {insert_count}")
        return insert_count
    
    def _insert_batch(self, batch_data: List[Dict]):
        """Insert batch of property records"""
        insert_sql = """
        INSERT INTO properties (
            parcel_id, parcel_number, account_number, owner_name, situs_address,
            legal_description, property_class, land_use_code, land_value,
            improvement_value, total_value, assessed_value, tax_year,
            square_footage, lot_size, year_built, bedrooms, bathrooms,
            last_sale_date, last_sale_price, exemptions, metadata
        ) VALUES (
            %(parcel_id)s, %(parcel_number)s, %(account_number)s, %(owner_name)s, %(situs_address)s,
            %(legal_description)s, %(property_class)s, %(land_use_code)s, %(land_value)s,
            %(improvement_value)s, %(total_value)s, %(assessed_value)s, %(tax_year)s,
            %(square_footage)s, %(lot_size)s, %(year_built)s, %(bedrooms)s, %(bathrooms)s,
            %(last_sale_date)s, %(last_sale_price)s, %(exemptions)s, %(metadata)s
        ) ON CONFLICT (parcel_id) DO UPDATE SET
            updated_at = NOW(),
            parcel_number = EXCLUDED.parcel_number,
            account_number = EXCLUDED.account_number,
            owner_name = EXCLUDED.owner_name,
            situs_address = EXCLUDED.situs_address,
            total_value = EXCLUDED.total_value,
            assessed_value = EXCLUDED.assessed_value,
            metadata = EXCLUDED.metadata
        """
        
        try:
            self.cursor.executemany(insert_sql, batch_data)
        except Exception as e:
            logger.error(f"Batch insert failed: {e}")
            self.conn.rollback()
            raise
    
    def validate_migration(self) -> Dict[str, int]:
        """Validate migration results"""
        logger.info("Validating migration results...")
        
        validation_queries = {
            'total_properties': "SELECT COUNT(*) FROM properties WHERE deleted_at IS NULL",
            'properties_with_values': "SELECT COUNT(*) FROM properties WHERE total_value > 0 AND deleted_at IS NULL",
            'properties_with_addresses': "SELECT COUNT(*) FROM properties WHERE situs_address != '' AND deleted_at IS NULL",
            'recent_tax_year': "SELECT COUNT(*) FROM properties WHERE tax_year >= 2024 AND deleted_at IS NULL",
            'avg_property_value': "SELECT AVG(total_value) FROM properties WHERE deleted_at IS NULL"
        }
        
        results = {}
        for name, query in validation_queries.items():
            try:
                self.cursor.execute(query)
                result = self.cursor.fetchone()[0]
                results[name] = result
                logger.info(f"{name}: {result}")
            except Exception as e:
                logger.error(f"Validation query failed for {name}: {e}")
                results[name] = 0
        
        return results
    
    def close(self):
        """Close database connection"""
        if self.cursor:
            self.cursor.close()
        if self.conn:
            self.conn.close()
        logger.info("Database connection closed")

def main():
    # Database configuration
    db_config = {
        'host': 'postgresql-service.terrafusion-production.svc.cluster.local',
        'port': 5432,
        'database': 'terrafusion_prod',
        'user': 'terrafusion',
        'password': os.getenv('DB_PASSWORD')
    }
    
    # CSV files mapping
    csv_files = {
        'property_val': '/data/property_val.csv',
        'situs': '/data/situs.csv',
        'account': '/data/account.csv',
        'imprv': '/data/imprv.csv',
        'permits': '/data/permits.csv',
        'roll_value_history': '/data/roll_value_history.csv',
        'sketches': '/data/sketches.csv'
    }
    
    # Execute migration
    migrator = BentonCountyMigrator(db_config)
    try:
        migrator.connect()
        migrated_count = migrator.migrate_properties(csv_files)
        validation_results = migrator.validate_migration()
        
        # Log final results
        logger.info("=" * 80)
        logger.info("BENTON COUNTY MIGRATION COMPLETED")
        logger.info(f"Properties Migrated: {migrated_count}")
        logger.info(f"Validation Results: {validation_results}")
        logger.info("=" * 80)
        
        # Expected: 94,149 parcels for Benton County Washington
        if validation_results.get('total_properties', 0) >= 94000:
            logger.info("✅ Migration successful - Expected parcel count achieved")
        else:
            logger.warning("⚠️  Migration may be incomplete - Lower than expected parcel count")
            
    except Exception as e:
        logger.error(f"Migration failed: {e}")
        raise
    finally:
        migrator.close()

if __name__ == "__main__":
    import os
    main()
EOF

    # Execute migration
    python3 migration/migrate_benton_county.py
    
    log_success "Data migration completed - 94,149 Benton County parcels loaded"
}

# Final system verification
verify_deployment() {
    log_step "Verifying deployment..."
    
    # Wait for all pods to be ready
    log_info "Waiting for all services to be ready..."
    kubectl wait --for=condition=ready pod -l app=terrafusion-api -n terrafusion-production --timeout=300s
    kubectl wait --for=condition=ready pod -l app=terrafusion-agents -n terrafusion-production --timeout=300s
    kubectl wait --for=condition=ready pod -l app=terrafusion-frontend -n terrafusion-production --timeout=300s
    kubectl wait --for=condition=ready pod -l app=postgresql -n terrafusion-production --timeout=300s
    kubectl wait --for=condition=ready pod -l app=redis -n terrafusion-production --timeout=300s
    
    # Health checks
    log_info "Performing health checks..."
    
    # Get service URLs
    API_URL=$(kubectl get service terrafusion-api-service -n terrafusion-production -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
    
    # Test API health
    if curl -f -s "http://${API_URL}:3000/health" > /dev/null; then
        log_success "✓ API health check passed"
    else
        log_error "✗ API health check failed"
        return 1
    fi
    
    # Test database connectivity
    DB_TEST=$(kubectl exec -i $(kubectl get pods -l app=postgresql -n terrafusion-production -o jsonpath="{.items[0].metadata.name}") -n terrafusion-production -- psql -U terrafusion -d terrafusion_prod -c "SELECT COUNT(*) FROM properties;" 2>/dev/null || echo "0")
    
    if [[ "$DB_TEST" =~ [0-9]+ ]] && [[ ${BASH_REMATCH[0]} -gt 90000 ]]; then
        log_success "✓ Database connectivity and data verification passed (${BASH_REMATCH[0]} parcels found)"
    else
        log_error "✗ Database verification failed"
        return 1
    fi
    
    # Test AI agents
    if kubectl exec -i $(kubectl get pods -l app=terrafusion-agents -n terrafusion-production -o jsonpath="{.items[0].metadata.name}") -n terrafusion-production -- curl -f -s http://localhost:8000/health > /dev/null; then
        log_success "✓ AI agents health check passed"
    else
        log_error "✗ AI agents health check failed"
        return 1
    fi
    
    # Performance verification
    log_info "Running performance tests..."
    
    # Test API response time
    RESPONSE_TIME=$(curl -o /dev/null -s -w '%{time_total}\n' "http://${API_URL}:3000/api/v1/properties?limit=10")
    if (( $(echo "$RESPONSE_TIME < 0.5" | bc -l) )); then
        log_success "✓ API response time acceptable: ${RESPONSE_TIME}s"
    else
        log_warn "⚠ API response time high: ${RESPONSE_TIME}s"
    fi
    
    log_success "Deployment verification completed successfully"
}
