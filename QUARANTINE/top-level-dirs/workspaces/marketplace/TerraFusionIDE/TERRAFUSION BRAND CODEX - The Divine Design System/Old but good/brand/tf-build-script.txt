@echo off
echo ========================================
echo Terrafusion OS - Build Script
echo ========================================
echo.

REM Clean previous builds
echo Cleaning previous builds...
if exist "Terrafusion.API\bin" rmdir /s /q "Terrafusion.API\bin"
if exist "Terrafusion.API\obj" rmdir /s /q "Terrafusion.API\obj"
if exist "Terrafusion.Shell\bin" rmdir /s /q "Terrafusion.Shell\bin"
if exist "Terrafusion.Shell\obj" rmdir /s /q "Terrafusion.Shell\obj"

REM Create directory structure
echo Creating directory structure...
if not exist "Terrafusion.API\wwwroot\css" mkdir "Terrafusion.API\wwwroot\css"
if not exist "Terrafusion.API\wwwroot\js" mkdir "Terrafusion.API\wwwroot\js"
if not exist "Terrafusion.API\wwwroot\icons" mkdir "Terrafusion.API\wwwroot\icons"
if not exist "Terrafusion.API\wwwroot\modules" mkdir "Terrafusion.API\wwwroot\modules"
if not exist "Terrafusion.API\Controllers" mkdir "Terrafusion.API\Controllers"
if not exist "Terrafusion.API\Services" mkdir "Terrafusion.API\Services"

REM Restore packages
echo.
echo Restoring NuGet packages...
cd Terrafusion.API
dotnet restore
cd ..

cd Terrafusion.Shell
dotnet restore
cd ..

REM Build API
echo.
echo Building Terrafusion API...
cd Terrafusion.API
dotnet build -c Release
if %errorlevel% neq 0 (
    echo ERROR: API build failed
    cd ..
    pause
    exit /b 1
)
cd ..

REM Build Shell
echo.
echo Building Terrafusion Shell...
cd Terrafusion.Shell
dotnet build -c Release
if %errorlevel% neq 0 (
    echo ERROR: Shell build failed
    cd ..
    pause
    exit /b 1
)
cd ..

REM Copy modules from championship
echo.
echo Copying modules...
if exist "..\championship\modules" (
    xcopy /E /I /Y "..\championship\modules\*" "Terrafusion.API\wwwroot\modules\"
    echo Modules copied successfully
) else (
    echo WARNING: Modules directory not found
    echo Expected: ..\championship\modules
)

REM Copy Benton County data if available
if exist "..\championship\src-tauri\benton_county_properties.json" (
    echo Copying Benton County data...
    copy /Y "..\championship\src-tauri\benton_county_properties.json" "Terrafusion.API\Data\"
)

REM Create release package
echo.
echo Creating release package...
if not exist "Release" mkdir "Release"

REM Copy executables
xcopy /Y "Terrafusion.Shell\bin\Release\net8.0-windows\*" "Release\"
xcopy /Y "Terrafusion.API\bin\Release\net8.0\*" "Release\"

REM Copy wwwroot
xcopy /E /I /Y "Terrafusion.API\wwwroot" "Release\wwwroot"

REM Create launch script for release
echo @echo off > "Release\Terrafusion.bat"
echo start /B Terrafusion.API.exe --environment Production >> "Release\Terrafusion.bat"
echo timeout /t 3 /nobreak ^>nul >> "Release\Terrafusion.bat"
echo start Terrafusion.Shell.exe >> "Release\Terrafusion.bat"

echo.
echo ========================================
echo Build Complete!
echo ========================================
echo.
echo Release package created in: Release\
echo.
echo To run Terrafusion:
echo   1. Navigate to Release\ folder
echo   2. Run Terrafusion.bat
echo.
pause
