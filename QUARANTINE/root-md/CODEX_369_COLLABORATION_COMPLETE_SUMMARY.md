## Database Schema Updates Required

### New Entity: NotificationPreferences

```sql
CREATE TABLE NotificationPreferences (
    Id INT PRIMARY KEY IDENTITY(1,1),
    UserId NVARCHAR(450) NOT NULL,

    -- Email Preferences
    EmailEnabled BIT NOT NULL DEFAULT 1,
    EmailAddress NVARCHAR(256),
    EmailCriticalAlerts BIT NOT NULL DEFAULT 1,
    EmailWarningAlerts BIT NOT NULL DEFAULT 1,
    EmailInfoAlerts BIT NOT NULL DEFAULT 0,
    EmailDailySummary BIT NOT NULL DEFAULT 1,
    EmailAchievements BIT NOT NULL DEFAULT 1,

    -- Slack Preferences
    SlackEnabled BIT NOT NULL DEFAULT 0,
    SlackWebhook NVARCHAR(500),
    SlackCriticalAlerts BIT NOT NULL DEFAULT 1,
    SlackWarningAlerts BIT NOT NULL DEFAULT 1,
    SlackInfoAlerts BIT NOT NULL DEFAULT 1,
    SlackStatusUpdates BIT NOT NULL DEFAULT 0,
    SlackDailySummary BIT NOT NULL DEFAULT 1,
    SlackAchievements BIT NOT NULL DEFAULT 1,

    -- Teams Preferences
    TeamsEnabled BIT NOT NULL DEFAULT 0,
    TeamsWebhook NVARCHAR(500),
    TeamsCriticalAlerts BIT NOT NULL DEFAULT 1,
    TeamsWarningAlerts BIT NOT NULL DEFAULT 1,
    TeamsInfoAlerts BIT NOT NULL DEFAULT 1,
    TeamsStatusUpdates BIT NOT NULL DEFAULT 0,
    TeamsDailySummary BIT NOT NULL DEFAULT 1,
    TeamsAchievements BIT NOT NULL DEFAULT 1,

    -- Alert Thresholds
    UltimatePowerScoreThreshold FLOAT NOT NULL DEFAULT 7.2,
    DomainScoreThreshold FLOAT NOT NULL DEFAULT 0.6,
    CriticalAlertThreshold FLOAT NOT NULL DEFAULT 7.2,
    WarningAlertThreshold FLOAT NOT NULL DEFAULT 9.6,

    -- Daily Summary Preferences
    DailySummaryEnabled BIT NOT NULL DEFAULT 1,
    DailySummaryTime NVARCHAR(10) DEFAULT '08:00',
    DailySummaryTimezone NVARCHAR(50) DEFAULT 'America/Los_Angeles',
    DailySummaryIncludeAlerts BIT NOT NULL DEFAULT 1,
    DailySummaryIncludePerformance BIT NOT NULL DEFAULT 1,
    DailySummaryIncludeTrends BIT NOT NULL DEFAULT 1,

    -- Audit fields
    CreatedAt DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME2 NOT NULL DEFAULT GETUTCDATE(),

    CONSTRAINT FK_NotificationPreferences_User FOREIGN KEY (UserId) REFERENCES AspNetUsers(Id) ON DELETE CASCADE,
    CONSTRAINT UQ_NotificationPreferences_UserId UNIQUE (UserId)
);

CREATE INDEX IX_NotificationPreferences_UserId ON NotificationPreferences(UserId);
```
