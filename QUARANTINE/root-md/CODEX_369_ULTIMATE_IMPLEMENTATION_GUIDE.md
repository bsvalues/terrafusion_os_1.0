    {
        if (rawCombinedValue >= DANGER_THRESHOLD)
        {
            return new SafeguardStatus

                SafetyMargin = 0,

        {
            return new SafeguardStatus
            {
                Level = "WARNING",
                SafeFromImbalance = true,
                SafetyMargin = DANGER_THRESHOLD - rawCombinedValue,
                Message = "⚠️ WARNING: Within 66 points of imbalance threshold",
                RecommendedAction = "Review and adjust domain weights"
            };
        }
        else if (rawCombinedValue >= SAFE_THRESHOLD)
        {
            return new SafeguardStatus
            {
                Level = "CAUTION",
                SafeFromImbalance = true,
                SafetyMargin = DANGER_THRESHOLD - rawCombinedValue,
                Message = "✓ CAUTION: Monitor domain balance",
                RecommendedAction = "Continue monitoring"
            };
        }
        else
        {
            return new SafeguardStatus
            {
                Level = "SAFE",
                SafeFromImbalance = true,
                SafetyMargin = DANGER_THRESHOLD - rawCombinedValue,
                Message = "✅ SAFE: Excellent balance maintained",
                RecommendedAction = "No action required"
            };
        }
    }
}
