export const baseApi = createApi({
  reducerPath: 'api',
  baseQuery: fetchBaseQuery({
    baseUrl: process.env.REACT_APP_API_GATEWAY || 'http://localhost:5002',
    prepareHeaders: (headers, { getState }) => {
      headers.set('Accept', 'application/vnd.terrafusion.tf+json;version=1');

      if (state.auth.token) {

      if (state.trust.score) {

      return headers;
    }
  }),

### Phase 3: Micro-Frontend Architecture (Week 5-6)
**Goal**: Modularize frontends with security

#### 3.1 Shell Application
```typescript
// frontend/shell/src/index.tsx
import { registerApplication, start } from 'single-spa';

// Manifest with integrity checking
async function loadMicroFrontend(name: string, url: string) {
  const manifest = await fetch(`/api/manifests/${name}`).then(r => r.json());
  
  // Verify signature
  if (!await verifySignature(manifest)) {
    throw new Error(`Invalid signature for ${name}`);
  }
  
  // Load with SRI
  return System.import(url);
}

registerApplication({
  name: '@terrafusion/desktop',
  app: () => loadMicroFrontend('desktop', 'http://localhost:5070/desktop.js'),
  activeWhen: ['/desktop']
});

registerApplication({
  name: '@terrafusion/command',
  app: () => loadMicroFrontend('command', 'http://localhost:5060/command.js'),
  activeWhen: ['/operations']
});

start();
```

### Phase 4: Performance & Security (Week 7-8)
**Goal**: Optimize and harden

#### 4.1 WASM Trust Calculator
```rust
// wasm/trust-calculator/src/lib.rs
#[wasm_bindgen]
pub struct TrustCalculator {
    weights: Vec<f64>,
}

#[wasm_bindgen]
impl TrustCalculator {
    pub fn calculate(&self, metrics: &[f64]) -> f64 {
        metrics.iter()
            .zip(&self.weights)
            .map(|(m, w)| m * w)
            .sum::<f64>()
            .clamp(0.0, 1.0)
    }
}
```

## Migration Strategy for Existing Frontends

### Desktop Environment (5070) Migration
```typescript
// Step 1: Wrap existing desktop in adapter
class DesktopAdapter {
  private legacyDesktop: LegacyDesktop;
  private newApi: typeof baseApi;
  
  async initialize() {
    // Keep all existing functionality
    this.legacyDesktop = new LegacyDesktop();
    
    // Gradually migrate to new API
    this.newApi = baseApi;
  }
  
  // Proxy pattern: new calls use new API, fallback to legacy
  async loadServices() {
    try {
      return await this.newApi.endpoints.getServices.query();
    } catch {
      return this.legacyDesktop.loadServices();
    }
  }
}
```
