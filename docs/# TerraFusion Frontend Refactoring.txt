## File 1: Enhanced Trust Fabric (Execute First)

```python
# services/trust-fabric-enhanced.py
import asyncio
import time
import json
import hashlib
from datetime import datetime
from typing import Dict, Optional
from aiohttp import web
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class EnhancedTrustFabric:
    def __init__(self):
        self.registrations: Dict[tuple, dict] = {}
        self.heartbeat_tasks = {}
        self.gc_interval = 10
        self.app = web.Application()
        self.setup_routes()
    
    def setup_routes(self):
        self.app.router.add_post('/api/trust-fabric/register', self.register_service)
        self.app.router.add_post('/api/trust-fabric/heartbeat/{instance_id}', self.heartbeat)
        self.app.router.add_get('/api/trust-fabric/services', self.get_services)
        self.app.router.add_delete('/api/trust-fabric/deregister/{instance_id}', self.deregister)
        self.app.router.add_get('/health', self.health_check)
    
    async def register_service(self, request: web.Request):
        data = await request.json()
        
        # Generate instance ID
        instance_id = f"{data['service_name']}-{int(time.time())}-{hashlib.md5(str(time.time()).encode()).hexdigest()[:8]}"
        
        # Dedupe by (service_name, port)
        service_key = (data['service_name'], data['port'])
        
        # Remove old registration if exists
        if service_key in self.registrations:
            old_instance = self.registrations[service_key]['instance_id']
            logger.info(f"Deduping service {service_key}: {old_instance} -> {instance_id}")
            if old_instance in self.heartbeat_tasks:
                self.heartbeat_tasks[old_instance].cancel()
        
        # Store registration
        self.registrations[service_key] = {
            'instance_id': instance_id,
            'service_name': data['service_name'],
            'port': data['port'],
            'version': data.get('version', '1.0.0'),
            'capabilities': data.get('capabilities', []),
            'trust_score': data.get('trust_score', 0.9),
            'lease_ttl_sec': data.get('lease_ttl_sec', 30),
            'last_heartbeat': time.time(),
            'registered_at': datetime.utcnow().isoformat()
        }
        
        # Start heartbeat monitor
        task = asyncio.create_task(self.monitor_heartbeat(instance_id, service_key))
        self.heartbeat_tasks[instance_id] = task
        
        logger.info(f"Registered service: {instance_id}")
        
        return web.json_response({
            'instance_id': instance_id,
            'heartbeat_interval': data.get('lease_ttl_sec', 30) // 3,
            'heartbeat_endpoint': f'/api/trust-fabric/heartbeat/{instance_id}'
        })
    
    async def heartbeat(self, request: web.Request):
        instance_id = request.match_info['instance_id']
        
        # Find registration by instance_id
        for key, reg in self.registrations.items():
            if reg['instance_id'] == instance_id:
                reg['last_heartbeat'] = time.time()
                return web.json_response({'status': 'ok', 'instance_id': instance_id})
        
        return web.json_response({'error': 'Instance not found'}, status=404)
    
    async def get_services(self, request: web.Request):
        services = []
        now = time.time()
        
        for (name, port), reg in self.registrations.items():
            # Only return active services
            if now - reg['last_heartbeat'] < reg['lease_ttl_sec'] * 2:
                services.append({
                    'service_id': reg['instance_id'],
                    'service_name': name,
                    'port': port,
                    'version': reg['version'],
                    'trust_score': reg['trust_score'],
                    'capabilities': reg['capabilities'],
                    'last_heartbeat': reg['last_heartbeat']
                })
        
        return web.json_response({
            'services': services,
            'count': len(services),
            'timestamp': datetime.utcnow().isoformat()
        })
    
    async def deregister(self, request: web.Request):
        instance_id = request.match_info['instance_id']
        
        for key, reg in list(self.registrations.items()):
            if reg['instance_id'] == instance_id:
                del self.registrations[key]
                if instance_id in self.heartbeat_tasks:
                    self.heartbeat_tasks[instance_id].cancel()
                logger.info(f"Deregistered service: {instance_id}")
                return web.json_response({'status': 'deregistered'})
        
        return web.json_response({'error': 'Instance not found'}, status=404)
    
    async def health_check(self, request: web.Request):
        return web.json_response({'status': 'healthy', 'services': len(self.registrations)})
    
    async def monitor_heartbeat(self, instance_id: str, service_key: tuple):
        """Monitor heartbeat and remove stale services"""
        while True:
            await asyncio.sleep(10)
            
            if service_key not in self.registrations:
                break
                
            reg = self.registrations[service_key]
            if time.time() - reg['last_heartbeat'] > reg['lease_ttl_sec'] * 2:
                logger.info(f"Service {instance_id} heartbeat expired, removing")
                del self.registrations[service_key]
                if instance_id in self.heartbeat_tasks:
                    del self.heartbeat_tasks[instance_id]
                break
    
    async def start_gc(self):
        """Garbage collector for stale services"""
        while True:
            await asyncio.sleep(self.gc_interval)
            now = time.time()
            
            to_remove = []
            for key, reg in self.registrations.items():
                if now - reg['last_heartbeat'] > reg['lease_ttl_sec'] * 2:
                    to_remove.append(key)
            
            for key in to_remove:
                instance_id = self.registrations[key]['instance_id']
                del self.registrations[key]
                if instance_id in self.heartbeat_tasks:
                    self.heartbeat_tasks[instance_id].cancel()
                    del self.heartbeat_tasks[instance_id]
                logger.info(f"GC: Removed stale service {key}")

async def main():
    fabric = EnhancedTrustFabric()
    
    # Start garbage collector
    asyncio.create_task(fabric.start_gc())
    
    runner = web.AppRunner(fabric.app)
    await runner.setup()
    site = web.TCPSite(runner, 'localhost', 5000)
    
    logger.info("Enhanced Trust Fabric starting on http://localhost:5000")
    await site.start()
    
    # Keep running
    while True:
        await asyncio.sleep(3600)

if __name__ == '__main__':
    asyncio.run(main())
```

```typescript
// services/gateway-v2/src/index.ts
import express from 'express';
import { createProxyMiddleware } from 'http-proxy-middleware';
import rateLimit from 'express-rate-limit';
import Redis from 'ioredis';
import { register, collectDefaultMetrics } from 'prom-client';
import winston from 'winston';
import { z } from 'zod';

// Logger setup
const logger = winston.createLogger({
  level: 'info',
  format: winston.format.json(),
  transports: [
    new winston.transports.Console({
      format: winston.format.simple()
    })
  ]
});

// Redis client
const redis = new Redis({
  host: 'localhost',
  port: 6379
});

// Express app
const app = express();
const PORT = process.env.PORT || 5002;

// Prometheus metrics
collectDefaultMetrics();

// Request ID middleware
app.use((req, res, next) => {
  req.id = `req-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  res.setHeader('X-Request-ID', req.id);
  next();
});

// Canary routing
app.use((req, res, next) => {
  const canaryPercentage = parseInt(process.env.CANARY_PERCENTAGE || '10');
  const isCanary = Math.random() * 100 < canaryPercentage;
  
  if (isCanary) {
    res.setHeader('X-Gateway', 'v2');
    res.setHeader('X-Canary', 'true');
    logger.info(`Canary request: ${req.id}`);
  }
  
  next();
});

// Rate limiting
const limiter = rateLimit({
  windowMs: 60 * 1000, // 1 minute
  max: 100,
  standardHeaders: true,
  legacyHeaders: false,
  handler: (req, res) => {
    res.status(429).json({
      error: 'Too many requests',
      retryAfter: 60
    });
  }
});

app.use('/api/', limiter);

// Version negotiation middleware
app.use((req, res, next) => {
  const acceptHeader = req.headers.accept || '';
  const versionMatch = acceptHeader.match(/version=(\d+)/);
  const requestedVersion = versionMatch ? parseInt(versionMatch[1]) : 1;
  
  req.apiVersion = requestedVersion;
  res.setHeader('X-API-Version', requestedVersion.toString());
  
  next();
});

// Circuit breaker state
const circuitBreakerState = new Map<string, {
  failures: number;
  lastFailTime: number;
  state: 'CLOSED' | 'OPEN' | 'HALF_OPEN';
}>();

// Circuit breaker middleware
const circuitBreaker = (serviceName: string) => {
  return (proxyReq: any, req: any, res: any) => {
    const state = circuitBreakerState.get(serviceName) || {
      failures: 0,
      lastFailTime: 0,
      state: 'CLOSED'
    };
    
    if (state.state === 'OPEN') {
      const timeSinceLastFail = Date.now() - state.lastFailTime;
      if (timeSinceLastFail > 60000) { // 1 minute
        state.state = 'HALF_OPEN';
      } else {
        res.status(503).json({
          error: 'Service temporarily unavailable',
          service: serviceName,
          retryAfter: 60 - Math.floor(timeSinceLastFail / 1000)
        });
        return;
      }
    }
  };
};

// Service routes with proxy
const services = {
  '/api/trust-fabric': 'http://localhost:5000',
  '/api/desktop': 'http://localhost:5070',
  '/api/command': 'http://localhost:5060',
  '/api/analytics': 'http://localhost:5050',
  '/api/datafusion': 'http://localhost:5250'
};

Object.entries(services).forEach(([path, target]) => {
  const serviceName = path.split('/').pop() || 'unknown';
  
  app.use(path, createProxyMiddleware({
    target,
    changeOrigin: true,
    onProxyReq: (proxyReq, req: any) => {
      proxyReq.setHeader('X-Request-ID', req.id);
      proxyReq.setHeader('X-Gateway-Version', '2');
      proxyReq.setHeader('X-API-Version', req.apiVersion);
      
      // Apply circuit breaker
      circuitBreaker(serviceName)(proxyReq, req, null);
    },
    onProxyRes: (proxyRes, req, res) => {
      // Track success/failure for circuit breaker
      const state = circuitBreakerState.get(serviceName) || {
        failures: 0,
        lastFailTime: 0,
        state: 'CLOSED'
      };
      
      if (proxyRes.statusCode && proxyRes.statusCode >= 500) {
        state.failures++;
        state.lastFailTime = Date.now();
        
        if (state.failures > 5) {
          state.state = 'OPEN';
          logger.error(`Circuit breaker OPEN for ${serviceName}`);
        }
      } else if (state.state === 'HALF_OPEN') {
        state.failures = 0;
        state.state = 'CLOSED';
        logger.info(`Circuit breaker CLOSED for ${serviceName}`);
      }
      
      circuitBreakerState.set(serviceName, state);
    },
    onError: (err, req, res: any) => {
      logger.error(`Proxy error for ${serviceName}:`, err);
      res.status(502).json({
        error: 'Bad gateway',
        service: serviceName,
        requestId: (req as any).id
      });
    }
  }));
});

// Health check
app.get('/health', (req, res) => {
  res.json({
    status: 'healthy',
    version: '2.0.0',
    uptime: process.uptime(),
    timestamp: new Date().toISOString()
  });
});

// Metrics endpoint
app.get('/metrics', async (req, res) => {
  res.set('Content-Type', register.contentType);
  res.end(await register.metrics());
});

// Start server
app.listen(PORT, () => {
  logger.info(`API Gateway v2 running on http://localhost:${PORT}`);
  logger.info(`Canary percentage: ${process.env.CANARY_PERCENTAGE || 10}%`);
});
```

## File 3: RTK Query Base Configuration

```typescript
// frontend/packages/shared/src/api/baseApi.ts
import { createApi, fetchBaseQuery } from '@reduxjs/toolkit/query/react';
import { z } from 'zod';

// Service schema validation
const ServiceSchema = z.object({
  service_id: z.string(),
  service_name: z.string(),
  port: z.number(),
  version: z.string(),
  trust_score: z.number().min(0).max(1),
  capabilities: z.array(z.string()),
  last_heartbeat: z.number()
});

const ServicesResponseSchema = z.object({
  services: z.array(ServiceSchema),
  count: z.number(),
  timestamp: z.string()
});

// Generate request ID
const generateRequestId = () => 
  `ui-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;

// Base API configuration
export const baseApi = createApi({
  reducerPath: 'api',
  baseQuery: fetchBaseQuery({
    baseUrl: process.env.REACT_APP_API_GATEWAY || 'http://localhost:5002',
    prepareHeaders: (headers, { getState }) => {
      // API versioning
      headers.set('Accept', 'application/vnd.terrafusion.tf+json;version=1');
      
      // Auth token
      const state = getState() as any;
      if (state.auth?.token) {
        headers.set('Authorization', `Bearer ${state.auth.token}`);
      }
      
      // Trust score
      if (state.trust?.score) {
        headers.set('X-Trust-Score', state.trust.score.toFixed(3));
      }
      
      // Request ID for tracing
      headers.set('X-Request-ID', generateRequestId());
      
      return headers;
    },
    validateStatus: (response, body) => {
      return response.status >= 200 && response.status < 300;
    }
  }),
  tagTypes: ['Service', 'Property', 'Tax', 'GIS', 'User'],
  endpoints: (builder) => ({
    // Service discovery
    getServices: builder.query({
      query: () => '/api/trust-fabric/services',
      transformResponse: (response: unknown) => {
        // Validate with Zod
        return ServicesResponseSchema.parse(response);
      },
      providesTags: ['Service']
    }),
    
    // Register service
    registerService: builder.mutation({
      query: (service) => ({
        url: '/api/trust-fabric/register',
        method: 'POST',
        body: service
      }),
      invalidatesTags: ['Service']
    }),
    
    // Heartbeat
    sendHeartbeat: builder.mutation({
      query: (instanceId: string) => ({
        url: `/api/trust-fabric/heartbeat/${instanceId}`,
        method: 'POST'
      })
    })
  })
});

export const {
  useGetServicesQuery,
  useRegisterServiceMutation,
  useSendHeartbeatMutation
} = baseApi;
```
