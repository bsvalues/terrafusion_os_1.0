class CitizenBehaviorTransformer(nn.Module):
    """Transformer model for citizen behavior prediction"""
    
    def __init__(self, input_dim=512, hidden_dim=1024, num_layers=8, num_heads=16):
        super().__init__()
        
        self.input_projection = nn.Linear(input_dim, hidden_dim)
        self.positional_encoding = PositionalEncoding(hidden_dim)
        
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim,
            nhead=num_heads,
            dim_feedforward=hidden_dim * 4,
            dropout=0.1,
            activation='gelu'
        )
        
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers)
        
        # Multi-task prediction heads
        self.service_demand_head = nn.Sequential(
            nn.Linear(hidden_dim, 256),
            nn.ReLU(),
            nn.Dropout(0.1),
            nn.Linear(256, 64),
            nn.ReLU(),
            nn.Linear(64, 10)  # 10 different services
        )
        
        self.compliance_head = nn.Sequential(
            nn.Linear(hidden_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 1),
            nn.Sigmoid()
        )
        
        self.satisfaction_head = nn.Sequential(
            nn.Linear(hidden_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 5)  # 5-point satisfaction scale
        )
        
    def forward(self, x):
        # Project input features
        x = self.input_projection(x)
        x = self.positional_encoding(x)
        
        # Transformer encoding
        encoded = self.transformer(x)
        
        # Global representation (mean pooling)
        global_repr = encoded.mean(dim=1)
        
        # Multi-task predictions
        service_demand = self.service_demand_head(global_repr)
        compliance = self.compliance_head(global_repr)
        satisfaction = self.satisfaction_head(global_repr)
        
        return {
            'service_demand': service_demand,
            'compliance_probability': compliance,
            'satisfaction_prediction': satisfaction,
            'embedding': global_repr
        }

class PolicyImpactSimulator(nn.Module):
    """Neural network for policy impact simulation"""
    
    def __init__(self):
        super().__init__()
        
        # Policy feature encoder
        self.policy_encoder = nn.Sequential(
            nn.Linear(256, 512),
            nn.ReLU(),
            nn.BatchNorm1d(512),
            nn.Dropout(0.2),
            nn.Linear(512, 256),
            nn.ReLU()
        )
        
        # Population feature encoder
        self.population_encoder = nn.Sequential(
            nn.Linear(512, 512),
            nn.ReLU(),
            nn.BatchNorm1d(512),
            nn.Dropout(0.2),
            nn.Linear(512, 256),
            nn.ReLU()
        )
        
        # Impact prediction heads
        self.economic_impact_head = nn.Sequential(
            nn.Linear(512, 128),
            nn.ReLU(),
            nn.Linear(128, 32),
            nn.ReLU(),
            nn.Linear(32, 5)  # 5 economic indicators
        )
        
        self.behavioral_impact_head = nn.Sequential(
            nn.Linear(512, 128),
            nn.ReLU(),
            nn.Linear(128, 32),
            nn.ReLU(),
            nn.Linear(32, 8)  # 8 behavioral indicators
        )
        
        self.success_probability_head = nn.Sequential(
            nn.Linear(512, 64),
            nn.ReLU(),
            nn.Linear(64, 1),
            nn.Sigmoid()
        )
        
    def forward(self, policy_features, population_features):
        # Encode features
        policy_encoded = self.policy_encoder(policy_features)
        population_encoded = self.population_encoder(population_features)
        
        # Combine features
        combined = torch.cat([policy_encoded, population_encoded], dim=1)
        
        # Predict impacts
        economic_impact = self.economic_impact_head(combined)
        behavioral_impact = self.behavioral_impact_head(combined)
        success_probability = self.success_probability_head(combined)
        
        return {
            'economic_impact': economic_impact,
            'behavioral_impact': behavioral_impact,
            'success_probability': success_probability
        }
