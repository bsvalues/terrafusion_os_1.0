using Microsoft.Extensions.Caching.Distributed;
using StackExchange.Redis;

namespace Terrafusion.API.Services
{
    public class OptimizedValuationService : IValuationService
    {
        private readonly IMemoryCache _l1Cache;
        private readonly IDistributedCache _l2Cache;
        private readonly IDatabase _redisDb;
        private readonly ILogger<OptimizedValuationService> _logger;
        
        public OptimizedValuationService(
            IMemoryCache l1Cache,
            IDistributedCache l2Cache,
            IConnectionMultiplexer redis,
            ILogger<OptimizedValuationService> logger)
        {
            _l1Cache = l1Cache;
            _l2Cache = l2Cache;
            _redisDb = redis.GetDatabase();
            _logger = logger;
        }

        public async Task<ValuationResult> CalculatePropertyValuation(
            string propertyId, 
            CancellationToken cancellationToken = default)
        {
            var stopwatch = Stopwatch.StartNew();
            var cacheKey = $"valuation:{propertyId}:{DateTime.UtcNow:yyyyMMdd}";
            
            // L1 Cache Check (Memory - ~1ms)
            if (_l1Cache.TryGetValue(cacheKey, out ValuationResult cached))
            {
                _logger.LogInformation("L1 Cache hit for {PropertyId} in {ElapsedMs}ms", 
                    propertyId, stopwatch.ElapsedMilliseconds);
                return cached;
            }
            
            // L2 Cache Check (Redis - ~5ms)
            var l2Cached = await _l2Cache.GetStringAsync(cacheKey, cancellationToken);
            if (!string.IsNullOrEmpty(l2Cached))
            {
                var result = JsonSerializer.Deserialize<ValuationResult>(l2Cached);
                
                // Populate L1 cache
                _l1Cache.Set(cacheKey, result, TimeSpan.FromMinutes(15));
                
                _logger.LogInformation("L2 Cache hit for {PropertyId} in {ElapsedMs}ms", 
                    propertyId, stopwatch.ElapsedMilliseconds);
                return result;
            }
            
            // Calculate fresh valuation with parallel AI processing
            var valuation = await CalculateOptimizedValuation(propertyId, cancellationToken);
            
            // Cache in both layers
            var serialized = JsonSerializer.Serialize(valuation);
            await _l2Cache.SetStringAsync(cacheKey, serialized, new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = TimeSpan.FromHours(4)
            }, cancellationToken);
            
            _l1Cache.Set(cacheKey, valuation, TimeSpan.FromMinutes(15));
            
            _logger.LogInformation("Fresh valuation calculated for {PropertyId} in {ElapsedMs}ms", 
                propertyId, stopwatch.ElapsedMilliseconds);
                
            return valuation;
        }
        
        private async Task<ValuationResult> CalculateOptimizedValuation(
            string propertyId, 
            CancellationToken cancellationToken)

                CalculateInsuranceValue(propertyId, cancellationToken),

                PropertyId = propertyId,

                CalculatedAt = DateTime.UtcNow,

                CacheStatus = "FRESH",

            };
        }
    }

ON comparable_sales USING GIST (location)

-- Optimize valuation calculation function
CREATE OR REPLACE FUNCTION calculate_property_valuation_optimized(
    p_property_id UUID
) RETURNS TABLE(
    market_value DECIMAL(12,2),
    assessed_value DECIMAL(12,2),
    confidence_score DECIMAL(3,2)
) AS $$
BEGIN
    -- Use materialized view for performance
    RETURN QUERY
    SELECT 
        pv.market_value,
        pv.assessed_value,
        pv.confidence_score
    FROM property_valuations_materialized pv
    WHERE pv.property_id = p_property_id
    AND pv.valuation_date = (
        SELECT MAX(valuation_date) 
        FROM property_valuations_materialized 
        WHERE property_id = p_property_id
    );
END;

$$ LANGUAGE plpgsql STABLE;

using System.Diagnostics;

namespace Terrafusion.API.Middleware
{
    public class PerformanceMonitoringMiddleware
    {
        private readonly RequestDelegate _next;
        private readonly ILogger<PerformanceMonitoringMiddleware> _logger;
        
        public PerformanceMonitoringMiddleware(
            RequestDelegate next,
            ILogger<PerformanceMonitoringMiddleware> logger)
        {
            _next = next;
            _logger = logger;
        }
        
        public async Task InvokeAsync(HttpContext context)
        {
            var stopwatch = Stopwatch.StartNew();
            
            // Add performance headers
            context.Response.OnStarting(() =>
            {
                context.Response.Headers.Add("X-Processing-Time", 
                    stopwatch.ElapsedMilliseconds.ToString());
                context.Response.Headers.Add("X-Server-Time", 
                    DateTime.UtcNow.ToString("yyyy-MM-ddTHH:mm:ss.fffZ"));
                return Task.CompletedTask;
            });
            
            await _next(context);
            
            stopwatch.Stop();
            
            // Log slow requests
            if (stopwatch.ElapsedMilliseconds > 1000)
            {
                _logger.LogWarning("Slow request detected: {Method} {Path} took {ElapsedMs}ms",
                    context.Request.Method,
                    context.Request.Path,
                    stopwatch.ElapsedMilliseconds);
            }
            
            // Performance budget alerts
            if (context.Request.Path.StartsWithSegments("/api/valuation") && 
                stopwatch.ElapsedMilliseconds > 500)
            {
                _logger.LogError("Valuation performance budget exceeded: {ElapsedMs}ms > 500ms target",
                    stopwatch.ElapsedMilliseconds);
            }
        }
    }
}

namespace Terrafusion.API.Services

            try
            {
                // Phase 1: Basic credential validation
                var userValidation = await ValidateUserCredentials(request.Username, request.Password);
                if (!userValidation.IsValid)
                {
                    await _auditLogger.LogFailedAuthentication(request.Username, "Invalid credentials");
                    return AuthenticationResult.Failed("Invalid credentials");
                }
                
                // Phase 2: Multi-Factor Authentication (FISMA requirement)
                var mfaValidation = await ValidateMultiFactorAuthentication(
                    userValidation.User, request.MfaToken
                );
                if (!mfaValidation.IsValid)
                {
                    await _auditLogger.LogFailedAuthentication(request.Username, "MFA failed");
                    return AuthenticationResult.Failed("Multi-factor authentication failed");
                }
                
                // Phase 3: Security clearance validation
                var clearanceValidation = await ValidateSecurityClearance(userValidation.User);
                if (!clearanceValidation.HasRequiredClearance)
                {
                    await _auditLogger.LogFailedAuthentication(request.Username, "Insufficient clearance");
                    return AuthenticationResult.Failed("Insufficient security clearance");
                }
                
                // Phase 4: Generate government-grade JWT token
                var jwtToken = await GenerateGovernmentJwtToken(

                await _auditLogger.LogSuccessfulAuthentication(
                    userValidation.User, request.IpAddress, stopwatch.ElapsedMilliseconds
                );
                
                return AuthenticationResult.Success(jwtToken, userValidation.User);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Government authentication failed for user {Username}", request.Username);
                await _auditLogger.LogAuthenticationError(request.Username, ex.Message);
                return AuthenticationResult.Failed("Authentication system error");
            }
        }
        
        private async Task<JwtSecurityToken> GenerateGovernmentJwtToken(
            GovernmentUser user, SecurityClearanceLevel clearanceLevel)
        {
            var tokenHandler = new JsonWebTokenHandler();
            var key = Encoding.ASCII.GetBytes(_configuration["Jwt:GovernmentSecretKey"]);
            
            var claims = new List<Claim>
            {
                new Claim(ClaimTypes.NameIdentifier, user.Id.ToString()),
                new Claim(ClaimTypes.Name, user.Username),
                new Claim(ClaimTypes.Email, user.Email),
                new Claim("agency", user.Agency),
                new Claim("department", user.Department),
                new Claim("clearance_level", clearanceLevel.ToString()),
                new Claim("government_user", "true"),
                new Claim("fisma_compliant", "true"),
                new Claim("fedramp_authorized", "true")
            };
            
            // Add role-based claims
            foreach (var role in user.Roles)
            {
                claims.Add(new Claim(ClaimTypes.Role, role));
            }
            
            var tokenDescriptor = new SecurityTokenDescriptor
            {
                Subject = new ClaimsIdentity(claims),
                Expires = DateTime.UtcNow.AddHours(8), // Government standard: 8 hours
                Issuer = "Terrafusion-Government",
                Audience = "Government-Users",
                SigningCredentials = new SigningCredentials(
                    new SymmetricSecurityKey(key),
                    SecurityAlgorithms.HmacSha256Signature)
            };
            
            var token = tokenHandler.CreateToken(tokenDescriptor);

        }
